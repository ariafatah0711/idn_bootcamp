### regexnya

```
^nova-compute\.log

"(GET|POST|PUT|DELETE) ([^ ]+) HTTP\/([0-9.]+)" status: (\d{3}) len: (\d+) time: ([0-9\.]+)

\.(\d{4}-\d{2}-\d{2}_\d{2}:\d{2}:\d{2}) (\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}\.\d{3}) (\d+) (INFO|DEBUG|ERROR|WARNING|CRITICAL) ([\w\.]+)(?: \[req-([a-f0-9\-]+)(?: [^\]]*)?\])?(?: \[instance: ([a-f0-9\-]+)\])?(?: (.+))?
```

## shorcut
```bash
# setup
cd /var/ossec/ruleset/decoders
echo > 0585-openstack-nova.xml && -R root:wazuh 0585-openstack-nova.xml
cd /var/ossec/ruleset/rules
echo > 1001-openstack-nova-rules.xml && -R root:wazuh 1001-openstack-nova-rules.xml

chown -R root:wazuh /var/ossec/ruleset/decoders/0585-openstack-nova.xml
chown -R root:wazuh /var/ossec/ruleset/rules/1001-openstack-nova-rules.xml

# shorcut for copy paste
echo > /var/ossec/ruleset/decoders/0585-openstack-nova.xml && nano /var/ossec/ruleset/decoders/0585-openstack-nova.xml
echo > /var/ossec/ruleset/rules/1001-openstack-nova-rules.xml && nano /var/ossec/ruleset/rules/1001-openstack-nova-rules.xml
```
